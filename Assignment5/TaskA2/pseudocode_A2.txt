Algorithm A2.1 - DispatchTotal (what b->total() does)

ALGORITHM DispatchTotal(b)
INPUT  : b, pointer to an object of a class derived from Bill
OUTPUT : the total of that object
BEGIN
 1.  table <- b->vptr          // hidden pointer set by constructor
 2.  fn <- table[slot(total)]  // slot fixed at compile time (1)
 3.  RETURN CALL fn(b)         // reaches the derived total()
END

Algorithm A2.4 - TotalByTypeTest (alternative concept)

ALGORITHM TotalByTypeTest(b)
INPUT  : b, pointer to Bill (total() is not virtual)
OUTPUT : the total of the object b points to
BEGIN
 1.  IF b IS-A OutPatientBill THEN
 2.      RETURN consultation + tests
 3.  ELSE IF b IS-A InPatientBill THEN
 4.      RETURN days * perDay + procedures + medicines
 5.  ELSE IF b IS-A DayCareBill THEN RETURN price
 6.  ELSE IF b IS-A EmergencyBill THEN
 7.      RETURN base + hours * rate
 8.  ELSE THROW BillingError("unknown bill type")
 9.  END IF
END
