Algorithm A3.1 - amountCovered of the three payers

ALGORITHM Insurance.amountCovered(bill)
INPUT  : bill; coverLimit, copayPercent of this Insurance
OUTPUT : amount paid by the insurer
BEGIN
 1.  payable <- bill->total() * (100 - copayPercent) / 100
 2.  IF payable > coverLimit THEN
 3.      THROW ClaimExceedsCoverError
 4.  END IF
 5.  RETURN payable
END

ALGORITHM GovernmentScheme.amountCovered(bill)
INPUT  : bill; rates, a table  package code -> fixed rate
OUTPUT : amount paid by the scheme
BEGIN
 1.  code <- bill->packageCode()
 2.  IF code NOT IN rates THEN THROW UnknownPackageError
 3.  RETURN MIN(rates[code], bill->total())
END
