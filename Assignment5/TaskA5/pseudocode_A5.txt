Algorithm A5.1 - catch, log, rethrow, recover, report

ALGORITHM SettleLogged(bill, payer)
BEGIN
 1.  TRY
 2.      RETURN Settle(bill, payer)
 3.  CATCH BillingError e
 4.      WRITE audit line (bill, payer, e.message)
 5.      RETHROW        // "throw;": same exception object
 6.  END TRY
END

ALGORITHM SettleOrFallback(bill, payer)
BEGIN
 1.  TRY
 2.      RETURN SettleLogged(bill, payer)
 3.  CATCH ClaimError e    // the claim is bad, the bill is not
 4.      WRITE recovery line (e.message)
 5.      RETURN Settle(bill, SelfPay)   // patient pays all
 6.  END TRY
END

ALGORITHM ProcessDay(bills, payer)
BEGIN
 1.  FOR EACH bill IN bills DO
 2.      TRY
 3.          s <- SettleOrFallback(bill, payer)
 4.          ADD s TO report
 5.      CATCH PatientRecordError e
 6.          REPORT e.message to the clerk
 7.          MARK bill "record missing"
 8.      CATCH BillingError e
 9.          REPORT e.message; MARK bill "failed"
10.      CATCH bad_alloc
11.          REPORT "out of memory"; MARK bill "retry later"
12.      CATCH ANY
13.          REPORT "unknown error"; MARK bill "failed"
14.      END TRY
15.  END FOR
END
